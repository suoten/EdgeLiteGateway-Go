1. Get the application URL by running these commands:
{{- if .Values.ingress.enabled }}
  export HOSTNAME=$(kubectl get ingress -n {{ .Release.Namespace }} -o jsonpath='{.items[0].spec.rules[0].host}')
  echo "Access the gateway at: http://$HOSTNAME"
{{- else }}
  export POD_NAME=$(kubectl get pods -n {{ .Release.Namespace }} -l "{{ include "edgelite.selectorLabels" . }}" -o jsonpath='{.items[0].metadata.name}')
  export CONTAINER_PORT={{ .Values.gateway.port }}
  kubectl port-forward pod/$POD_NAME $CONTAINER_PORT:$CONTAINER_PORT -n {{ .Release.Namespace }}
  echo "Access the gateway at: http://127.0.0.1:$CONTAINER_PORT"
{{- end }}
